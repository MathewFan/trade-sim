cmake_minimum_required(VERSION 3.20)
project(tradesim VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
option(TRADESIM_ASAN "Address and undefined behavior sanitizers" OFF)
option(TRADESIM_TSAN "Thread sanitizer" OFF)
if(TRADESIM_ASAN AND TRADESIM_TSAN)
  message(FATAL_ERROR "Choose ASAN or TSAN, not both")
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(TRADESIM_ASAN)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
  if(TRADESIM_TSAN)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  endif()
endif()
add_library(tradesim engine/engine.cpp)
target_include_directories(tradesim PUBLIC engine)
target_link_libraries(tradesim PUBLIC Threads::Threads)
foreach(app sharded_driver engine_mvp ws_server wal_verify wal_replay)
  add_executable(${app} apps/${app}.cpp)
  target_link_libraries(${app} PRIVATE tradesim)
endforeach()
set_target_properties(ws_server PROPERTIES OUTPUT_NAME ws_tradesim)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE tradesim)
add_test(NAME core_invariants COMMAND core_tests)
set_tests_properties(core_invariants PROPERTIES TIMEOUT 30)

add_executable(recovery_tests tests/recovery_tests.cpp)
target_link_libraries(recovery_tests PRIVATE tradesim)
add_test(NAME journal_recovery COMMAND recovery_tests)
set_tests_properties(journal_recovery PROPERTIES TIMEOUT 30)

add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE tradesim)
add_test(NAME websocket_protocol COMMAND protocol_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME websocket_loopback COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/websocket_integration.py $<TARGET_FILE:ws_server>)
  set_tests_properties(websocket_loopback PROPERTIES TIMEOUT 30)
endif()
